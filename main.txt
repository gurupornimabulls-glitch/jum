def value_fn_estimate(env,episodes,gamma = 1.0):
  num_states = env.observation_space.n
  num_actions = env.action_space.n
  V_pi = np.zeros(num_states)

  Q_pi = np.zeros((num_states,num_actions))
  returns_V = {s:[] for s in range(num_states)}
  returns_Q = {(s,a):[] for s in range(num_states) for a in range(num_actions)}
  for episode in episodes:
    states = episode["states"]
    actions = episode["actions"]
    rewards = episode["rewards"]
    T = len(rewards)
    G = 0.0
    returns=np.zeros(T)
    for t in reversed(range(T)):
      G = rewards[t]+gamma*G
      returns[t]=G
    visited_states,visited_actions = set(),set()
    for t in range(T):
      s,a = states[t],actions[t]
      if s not in visited_states:
        visited_states.add(s)
        returns_V[t].append(returns[t])
      if (s,a) not in visited_actions:
        visited_actions.add((s,a))
        returns_Q[(s,a)].append(returns[t])
  for s in range(num_states):
    if len(returns_V[s])>0:
      V_pi[s]=np.mean(returns_V[s])
    for a in range(num_actions):
      if len(returns_Q[(s,a)]):
        Q_pi[(s,a)]=np.mean(returns_Q[(s,a)])
  return V_pi,Q_pi

def choose_action(Q,s,epsilon):
  u = np.random.uniform(0,1)
  num_actions = Q.shape[1]
  if u<1-epsilon:
    return np.argmax(Q[s,:])
  else:
    return np.random.choice(num_actions)
